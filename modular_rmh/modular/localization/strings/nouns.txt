# Существительные с закрытым множеством значений — то, что возвращает parse_zone()
# в code/__HELPERS/medical.dm. Шаблон просит у них падеж через $N|acc и т.п.
#
# Формат:  английское = им. | род. | дат. | вин. | твор. | предл.
#
# Строки, начинающиеся с #, и пустые строки игнорируются.
# Разделитель полей — вертикальная черта, пробелы вокруг срезаются.

left arm   = левая рука    | левой руки    | левой руке    | левую руку    | левой рукой    | левой руке
right arm  = правая рука   | правой руки   | правой руке   | правую руку   | правой рукой   | правой руке
left hand  = левая кисть   | левой кисти   | левой кисти   | левую кисть   | левой кистью   | левой кисти
right hand = правая кисть  | правой кисти  | правой кисти  | правую кисть  | правой кистью  | правой кисти
left leg   = левая нога    | левой ноги    | левой ноге    | левую ногу    | левой ногой    | левой ноге
right leg  = правая нога   | правой ноги   | правой ноге   | правую ногу   | правой ногой   | правой ноге
left foot  = левая ступня  | левой ступни  | левой ступне  | левую ступню  | левой ступнёй  | левой ступне
right foot = правая ступня | правой ступни | правой ступне | правую ступню | правой ступнёй | правой ступне
left eye   = левый глаз    | левого глаза  | левому глазу  | левый глаз    | левым глазом   | левом глазу
right eye  = правый глаз   | правого глаза | правому глазу | правый глаз   | правым глазом  | правом глазу
chest      = грудь         | груди         | груди         | грудь         | грудью         | груди
head       = голова        | головы        | голове        | голову        | головой        | голове
throat     = горло         | горла         | горлу         | горло         | горлом         | горле
groin      = пах           | паха          | паху          | пах           | пахом          | паху
ear        = ухо           | уха           | уху           | ухо           | ухом           | ухе
nose       = нос           | носа          | носу          | нос           | носом          | носу
skull      = череп         | черепа        | черепу        | череп         | черепом        | черепе
mouth      = рот           | рта           | рту           | рот           | ртом           | рту

# Не часть тела, а местоимение. Бой в первом лице собирается как
# "[X] punches me in the chest!" — "me" стоит ровно там, где шаблон просит
# падеж, так что дешевле склонять его тем же механизмом, чем плодить
# отдельную пару паттернов на каждый глагол.
me         = я             | меня          | мне           | меня          | мной           | мне
